rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Users collection
    match /users/{userId} {
      allow read, write: if request.auth != null;
    }
    
    // Customers linked to an owner
    match /customers/{customerId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null && resource.data.ownerId == request.auth.uid;
    }
    
    // Tubewells
    match /tubewells/{tubewellId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null && resource.data.ownerId == request.auth.uid;
    }
    
    // Water usage
    match /water_usage/{usageId} {
      allow read, write: if request.auth != null;
    }
    
    // Queues
    match /queues/{queueId} {
      allow read, write: if request.auth != null;
    }
    
    // Customer links
    match /customer_links/{linkId} {
      allow read, write: if request.auth != null;
    }
    
    // Link requests
    match /link_requests/{requestId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null && (resource.data.ownerUid == request.auth.uid || resource.data.customerUid == request.auth.uid);
    }
    
    // Notifications
    match /notifications/{notifId} {
      allow read: if request.auth != null && resource.data.toUid == request.auth.uid;
      allow create: if request.auth != null;
      allow update: if request.auth != null && resource.data.toUid == request.auth.uid;
    }
    
    match /notifications/{notifId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null && (resource.data.toUid == request.auth.uid || request.auth.uid == resource.data.toUid);
    }
    
        match /customer_links/{linkId} {
      allow read: if request.auth != null;
      allow create, delete: if request.auth != null;
      allow update: if request.auth != null;
    }
  }
}
